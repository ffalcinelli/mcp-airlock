1//! # URL security policy
2//!
3//! - Remote endpoints (MCP server, authorization server) must use HTTPS, except on
4//!   loopback hosts or when the user explicitly allows insecure HTTP.
5//! - The OAuth redirect URI must be a loopback `http` URI (RFC 8252 §7.3), so the
6//!   callback server is never reachable from the network.
7
8use crate::Result;
9use anyhow::{bail, Context};
10use std::net::IpAddr;
11use tracing::warn;
12use url::{Host, Url};
13
14/// `localhost`, `127.0.0.0/8` or `::1`.
15pub(crate) fn is_loopback(url: &Url) -> bool {
16    match url.host() {
17        Some(Host::Domain(d)) => d.eq_ignore_ascii_case("localhost"),
18        Some(Host::Ipv4(ip)) => IpAddr::V4(ip).is_loopback(),
19        Some(Host::Ipv6(ip)) => IpAddr::V6(ip).is_loopback(),
20        None => false,
21    }
22}
23
24/// Requires `https`, or `http` on a loopback host. With `allow_insecure`, plain
25/// `http` anywhere is accepted with a warning.
26pub(crate) fn require_secure_url(url: &str, what: &str, allow_insecure: bool) -> Result<()> {
27    let parsed = Url::parse(url).with_context(|| format!("Invalid {what} '{url}'"))?;
28    match parsed.scheme() {
29        "https" => Ok(()),
30        "http" if is_loopback(&parsed) => Ok(()),
31        "http" if allow_insecure => {
32            warn!(
33                "{} '{}' uses plain HTTP; tokens may be exposed (--allow-insecure-http).",
34                what, url
35            );
36            Ok(())
37        }
38        "http" => bail!(
39            "{what} '{url}' must use HTTPS (plain HTTP is only allowed for loopback hosts, \
40             or with --allow-insecure-http)"
41        ),
42        other => bail!("{what} '{url}' has unsupported scheme '{other}'"),
43    }
44}
45
46/// Requires an `http` redirect URI on a loopback host with an explicit port.
47pub(crate) fn require_loopback_redirect(url: &str) -> Result<Url> {
48    let parsed = Url::parse(url).with_context(|| format!("Invalid redirect URL '{url}'"))?;
49    if parsed.scheme() != "http" || !is_loopback(&parsed) {
50        bail!(
51            "Redirect URL '{url}' must be an http URL on 127.0.0.1, [::1] or localhost \
52             (RFC 8252 §7.3)"
53        );
54    }
55    if parsed.port().is_none() {
56        bail!("Redirect URL '{url}' must include a port");
57    }
58    Ok(parsed)
59}

61#[cfg(test)]
62mod tests {
63    use super::*;
64
65    #[test]
66    fn test_is_loopback() {
67        for u in [
68            "http://localhost:1",
69            "http://LOCALHOST",
70            "http://127.0.0.1",
71            "http://127.1.2.3",
72            "http://[::1]:8080",
73        ] {
74            assert!(is_loopback(&Url::parse(u).unwrap()), "{u}");
75        }
76        for u in [
77            "http://example.com",
78            "http://10.0.0.1",
79            "http://localhost.example.com",
80            "http://0.0.0.0",
81        ] {
82            assert!(!is_loopback(&Url::parse(u).unwrap()), "{u}");
83        }
84    }
85
86    #[test]
87    fn test_require_secure_url() {
88        assert!(require_secure_url("https://mcp.example.com/mcp", "x", false).is_ok());
89        assert!(require_secure_url("http://127.0.0.1:8081/rpc", "x", false).is_ok());
90        assert!(require_secure_url("http://localhost/rpc", "x", false).is_ok());
91        let err = require_secure_url("http://mcp.example.com/mcp", "--remote-mcp-url", false)
92            .unwrap_err()
93            .to_string();
94        assert!(err.contains("must use HTTPS"), "{err}");
95        assert!(require_secure_url("http://mcp.example.com/mcp", "x", true).is_ok());
96        assert!(require_secure_url("ftp://mcp.example.com", "x", true).is_err());
97        assert!(require_secure_url("not a url", "x", true).is_err());
98    }
99
100    #[test]
101    fn test_require_loopback_redirect() {
102        assert!(require_loopback_redirect("http://127.0.0.1:8082/callback").is_ok());
103        assert!(require_loopback_redirect("http://localhost:8082/callback").is_ok());
104        assert!(require_loopback_redirect("http://[::1]:8082/callback").is_ok());
105        assert!(require_loopback_redirect("http://0.0.0.0:8082/callback").is_err());
106        assert!(require_loopback_redirect("http://192.168.1.2:8082/callback").is_err());
107        assert!(require_loopback_redirect("https://127.0.0.1:8082/callback").is_err());
108        assert!(require_loopback_redirect("http://127.0.0.1/callback").is_err());
109        assert!(require_loopback_redirect("r").is_err());
110    }
