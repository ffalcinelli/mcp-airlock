1//! # `WWW-Authenticate` parsing
2//!
3//! A small tokenizer for the challenge grammar of RFC 9110 §11.6.1:
4//!
5//! ```text
6//! challenge  = auth-scheme [ 1*SP ( token68 / #auth-param ) ]
7//! auth-param = token BWS "=" BWS ( token / quoted-string )
8//! ```
9//!
10//! Parameter names are matched whole and case-insensitively, quoted strings
11//! honour `\` escapes, and several challenges (in one header or across several
12//! header lines) are supported.
13
14use reqwest::header::{HeaderMap, WWW_AUTHENTICATE};
15use tracing::debug;
16
17/// A single authentication challenge, e.g. `DPoP error="use_dpop_nonce"`.
18#[derive(Debug, Default, Clone, PartialEq, Eq)]
19pub(crate) struct Challenge {
20    pub scheme: String,
21    pub params: Vec<(String, String)>,
22}
23
24impl Challenge {
25    pub fn param(&self, name: &str) -> Option<&str> {
26        self.params
27            .iter()
28            .find(|(k, _)| k.eq_ignore_ascii_case(name))
29            .map(|(_, v)| v.as_str())
30    }
31}
32
33/// The parameters of the `WWW-Authenticate` header(s) that mcp-airlock acts on.
34#[derive(Debug, Default)]
35pub(crate) struct WwwAuthenticate {
36    pub challenges: Vec<Challenge>,
37}
38
39impl WwwAuthenticate {
40    pub fn parse(headers: &HeaderMap) -> Self {
41        let mut challenges = Vec::new();
42        for value in headers.get_all(WWW_AUTHENTICATE) {
43            match value.to_str() {
44                Ok(v) => {
45                    debug!("Parsing WWW-Authenticate: {}", v);
46                    challenges.extend(parse_challenges(v));
47                }
48                Err(_) => debug!("Ignoring non-ASCII WWW-Authenticate header"),
49            }
50        }
51        Self { challenges }
52    }
53
54    /// The first value of `name` across all challenges.
55    pub fn param(&self, name: &str) -> Option<&str> {
56        self.challenges.iter().find_map(|c| c.param(name))
57    }
58
59    pub fn resource_metadata(&self) -> Option<&str> {
60        self.param("resource_metadata")
61    }
62
63    pub fn error(&self) -> Option<&str> {
64        self.param("error")
65    }
66
67    /// Whether any challenge carries `error=<code>`.
68    pub fn has_error(&self, code: &str) -> bool {
69        self.challenges
70            .iter()
71            .any(|c| c.param("error") == Some(code))
72    }
73
74    /// The space-separated `scope` parameter, split into scopes.
75    pub fn scope(&self) -> Option<Vec<String>> {
76        self.param("scope")
77            .map(|s| s.split_whitespace().map(str::to_string).collect())
78    }
79}

81fn is_ws(b: u8) -> bool {
82    b == b' ' || b == b'\t'
83}
84
85fn skip_ws(b: &[u8], mut i: usize) -> usize {
86    while i < b.len() && is_ws(b[i]) {
87        i += 1;
88    }
89    i
90}
91
92/// Parses one header value into its challenges. Malformed input stops the
93/// parse; whatever was parsed before is kept.
94pub(crate) fn parse_challenges(header: &str) -> Vec<Challenge> {
95    let b = header.as_bytes();
96    let mut out: Vec<Challenge> = Vec::new();
97    let mut i = 0;
98
99    loop {
100        while i < b.len() && (is_ws(b[i]) || b[i] == b',') {
101            i += 1;
102        }
103        if i >= b.len() {
104            break;
105        }
106
107        let start = i;
108        while i < b.len() && !is_ws(b[i]) && b[i] != b',' && b[i] != b'=' {
109            i += 1;
110        }
111        let token = &header[start..i];
112
113        let j = skip_ws(b, i);
114        if token.is_empty() || j >= b.len() || b[j] != b'=' {
115            // A bare token starts a new challenge (or is a token68, which we ignore).
116            if !token.is_empty() {
117                out.push(Challenge {
118                    scheme: token.to_string(),
119                    params: Vec::new(),
120                });
121            } else {
122                i += 1;
123            }
124            continue;
125        }
126
127        let mut j = skip_ws(b, j + 1);
128        let value = if j < b.len() && b[j] == b'"' {
129            j += 1;
130            let mut value = String::new();
131            let mut closed = false;
132            let mut seg = j;
133            while j < b.len() {
134                match b[j] {
135                    b'\\' if j + 1 < b.len() => {
136                        value.push_str(&header[seg..j]);
137                        let c = header[j + 1..].chars().next().unwrap_or_default();
138                        value.push(c);
139                        j += 1 + c.len_utf8();
140                        seg = j;
141                    }
142                    b'"' => {
143                        value.push_str(&header[seg..j]);
144                        j += 1;
145                        closed = true;
146                        break;
147                    }
148                    _ => j += 1,
149                }
150            }
151            if !closed {
152                debug!("Unterminated quoted string in WWW-Authenticate");
153                break;
154            }
155            value
156        } else {
157            let vs = j;
158            while j < b.len() && b[j] != b',' && !is_ws(b[j]) {
159                j += 1;
160            }
161            header[vs..j].to_string()
162        };
163        i = j;
164
165        let param = (token.to_ascii_lowercase(), value);
166        match out.last_mut() {
167            Some(c) => c.params.push(param),
168            None => out.push(Challenge {
169                scheme: String::new(),
170                params: vec![param],
171            }),
172        }
173    }
174    out
175}

177#[cfg(test)]
178mod tests {
179    use super::*;
180    use reqwest::header::HeaderValue;

182    fn parse(v: &'static str) -> WwwAuthenticate {
183        let mut headers = HeaderMap::new();
184        headers.insert(WWW_AUTHENTICATE, HeaderValue::from_static(v));
185        WwwAuthenticate::parse(&headers)
186    }
187
188    fn scope(v: &'static str) -> Option<Vec<String>> {
189        parse(v).scope()
190    }
191
192    fn strings(v: &[&str]) -> Option<Vec<String>> {
193        Some(v.iter().map(|s| s.to_string()).collect())
194    }

197    fn test_parse_empty_headers() {
198        let auth = WwwAuthenticate::parse(&HeaderMap::new());
199        assert_eq!(auth.resource_metadata(), None);
200        assert_eq!(auth.scope(), None);
201        assert_eq!(auth.error(), None);

288        // `error_description` must not be mistaken for `error`.
289        let auth = parse("Bearer error_description=\"error=bad\", error=\"invalid_token\"");
290        assert_eq!(auth.error(), Some("invalid_token"));
291        // A parameter name inside a quoted value is not a parameter.
