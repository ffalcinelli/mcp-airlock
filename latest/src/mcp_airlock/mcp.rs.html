1//! # MCP protocol eras and Streamable HTTP request metadata
2//!
3//! mcp-airlock forwards whatever the stdio client speaks:
4//!
5//! - **Modern** (2026-07-28 and later): every request carries its protocol
6//!   version in `_meta`; there is no `initialize`, no session and no GET stream.
7//! - **Legacy** (2025-11-25 and earlier): an `initialize` handshake negotiates
8//!   the version, which then goes in the `MCP-Protocol-Version` header.
9//!
10//! For HTTP, modern requests also mirror body fields into headers
11//! (`Mcp-Method`, `Mcp-Name`) so that intermediaries can route on them.
12
13use base64::{engine::general_purpose::STANDARD, Engine as _};
14use serde_json::Value;
15
16/// `_meta` key carrying the protocol version of a modern request.
17pub(crate) const META_PROTOCOL_VERSION: &str = "io.modelcontextprotocol/protocolVersion";
18
19/// JSON-RPC error code for headers that don't match the body (2026-07-28).
20pub(crate) const HEADER_MISMATCH: i64 = -32020;
21
22/// Which generation of the protocol a message belongs to.
23#[derive(Clone, Copy, Debug, PartialEq, Eq)]
24pub(crate) enum Era {
25    /// Per-request metadata (2026-07-28 and later).
26    Modern,
27    /// `initialize`-based sessions (2025-11-25 and earlier).
28    Legacy,
29}
30
31/// The protocol version declared in a modern request's `_meta`.
32pub(crate) fn meta_protocol_version(payload: &Value) -> Option<&str> {
33    payload
34        .get("params")?
35        .get("_meta")?
36        .get(META_PROTOCOL_VERSION)?
37        .as_str()
38}
39
40pub(crate) fn era_of(payload: &Value) -> Era {
41    if meta_protocol_version(payload).is_some() {
42        Era::Modern
43    } else {
44        Era::Legacy
45    }
46}
47
48pub(crate) fn method_of(payload: &Value) -> Option<&str> {
49    payload.get("method")?.as_str()
50}

52/// Whether `value` can go in an HTTP header as is: visible ASCII, space or tab,
53/// no leading/trailing whitespace, and not mistakable for an encoded value.
54fn is_plain_header_safe(value: &str) -> bool {
55    let bytes = value.as_bytes();
56    let printable = bytes
57        .iter()
58        .all(|&b| b == b'\t' || (0x20..=0x7e).contains(&b));
59    let trimmed =
60        !matches!(bytes.first(), Some(b' ' | b'\t')) && !matches!(bytes.last(), Some(b' ' | b'\t'));
61    let sentinel = value.starts_with("=?base64?") && value.ends_with("?=");
62    printable && trimmed && !sentinel
63}
64
65/// Encodes a value for an `Mcp-Name` / `Mcp-Param-*` header, using the
66/// `=?base64?…?=` form when it can't be sent as plain ASCII.
67pub(crate) fn encode_header_value(value: &str) -> String {
68    if is_plain_header_safe(value) {
69        value.to_string()
70    } else {
71        format!("=?base64?{}?=", STANDARD.encode(value.as_bytes()))
72    }
73}

75/// The standard request headers of a POST: `Mcp-Method` always, and
76/// `Mcp-Name` for `tools/call`, `prompts/get` and `resources/read`.
77pub(crate) fn standard_headers(payload: &Value) -> Vec<(&'static str, String)> {
78    let mut headers = Vec::new();
79    let Some(method) = method_of(payload) else {
80        return headers;
81    };
82    headers.push(("Mcp-Method", encode_header_value(method)));
83
84    let name_field = match method {
85        "tools/call" | "prompts/get" => Some("name"),
86        "resources/read" => Some("uri"),
87        _ => None,
88    };
89    if let Some(name) = name_field
90        .and_then(|f| payload.get("params")?.get(f))
91        .and_then(Value::as_str)
92    {
93        headers.push(("Mcp-Name", encode_header_value(name)));
94    }
95    headers
96}

98/// The `x-mcp-header` annotations of one tool: property path → header name
99/// (the `{name}` of `Mcp-Param-{name}`).
100pub(crate) type ToolHeaders = Vec<(Vec<String>, String)>;
101
102/// RFC 9110 `tchar`.
103fn is_tchar(c: char) -> bool {
104    c.is_ascii_alphanumeric() || "!#$%&'*+-.^_`|~".contains(c)
105}
106
107/// Whether a schema `type` is a primitive allowed to carry `x-mcp-header`:
108/// `string`, `integer` or `boolean` (optionally together with `null`).
109fn is_header_type(ty: Option<&Value>) -> bool {
110    let allowed = |t: &str| matches!(t, "string" | "integer" | "boolean");
111    match ty {
112        Some(Value::String(t)) => allowed(t),
113        Some(Value::Array(ts)) => {
114            let non_null: Vec<&str> = ts
115                .iter()
116                .filter_map(Value::as_str)
117                .filter(|t| *t != "null")
118                .collect();
119            ts.iter().all(Value::is_string) && non_null.len() == 1 && allowed(non_null[0])
120        }
121        _ => false,
122    }
123}

125/// Collects `x-mcp-header` annotations of the schema `node` at `path`.
126/// `reachable` is true while the path consists only of `properties` keys.
127fn collect_headers(
128    node: &Value,
129    path: &mut Vec<String>,
130    reachable: bool,
131    out: &mut ToolHeaders,
132) -> Result<(), String> {
133    match node {
134        Value::Object(map) => {
135            if let Some(annotation) = map.get("x-mcp-header") {
136                let name = annotation.as_str().ok_or("x-mcp-header must be a string")?;
137                if name.is_empty() || !name.chars().all(is_tchar) {
138                    return Err(format!("x-mcp-header '{name}' is not a valid header name"));
139                }
140                if !reachable || path.is_empty() {
141                    return Err(format!(
142                        "x-mcp-header '{name}' is not on a property reachable through `properties` only"
143                    ));
144                }
145                if !is_header_type(map.get("type")) {
146                    return Err(format!(
147                        "x-mcp-header '{name}' is on a parameter that is not a string, integer or boolean"
148                    ));
149                }
150                if out.iter().any(|(_, n)| n.eq_ignore_ascii_case(name)) {
151                    return Err(format!("x-mcp-header '{name}' is not unique"));
152                }
153                out.push((path.clone(), name.to_string()));
154            }
155            for (key, value) in map {
156                match (key.as_str(), value) {
157                    ("x-mcp-header", _) => {}
158                    ("properties", Value::Object(props)) => {
159                        for (prop, schema) in props {
160                            path.push(prop.clone());
161                            let res = collect_headers(schema, path, reachable, out);
162                            path.pop();
163                            res?;
164                        }
165                    }
166                    // Anything else (items, oneOf, $defs, if/then, ...) is not
167                    // statically reachable.
168                    _ => collect_headers(value, path, false, out)?,
169                }
170            }
171        }
172        Value::Array(items) => {
173            for item in items {
174                collect_headers(item, path, false, out)?;
175            }
176        }
177        _ => {}
178    }
179    Ok(())
180}

182/// Validates the `x-mcp-header` annotations of a tool's `inputSchema` and
183/// returns them, or why the tool definition is invalid.
184pub(crate) fn tool_header_annotations(input_schema: &Value) -> Result<ToolHeaders, String> {
185    let mut out = Vec::new();
186    collect_headers(input_schema, &mut Vec::new(), true, &mut out)?;
187    Ok(out)
188}

190/// The `Mcp-Param-*` headers of a `tools/call` request, from the arguments at
191/// the annotated paths. Missing and `null` values are omitted.
192pub(crate) fn param_headers(
193    arguments: Option<&Value>,
194    headers: &ToolHeaders,
195) -> Vec<(String, String)> {
196    let Some(arguments) = arguments else {
197        return Vec::new();
198    };
199    headers
200        .iter()
201        .filter_map(|(path, name)| {
202            let value = path.iter().try_fold(arguments, |v, key| v.get(key))?;
203            let text = match value {
204                Value::String(s) => s.clone(),
205                Value::Bool(b) => b.to_string(),
206                Value::Number(n) if n.is_i64() || n.is_u64() => n.to_string(),
207                _ => return None,
208            };
209            Some((format!("Mcp-Param-{name}"), encode_header_value(&text)))
210        })
211        .collect()
212}

219    #[test]
220    fn test_era_detection() {
221        let modern = json!({"jsonrpc": "2.0", "id": 1, "method": "tools/list",
222            "params": {"_meta": {META_PROTOCOL_VERSION: "2026-07-28"}}});
223        assert_eq!(era_of(&modern), Era::Modern);
224        assert_eq!(meta_protocol_version(&modern), Some("2026-07-28"));
225
226        let legacy = json!({"jsonrpc": "2.0", "id": 1, "method": "initialize",
227            "params": {"protocolVersion": "2025-11-25"}});
228        assert_eq!(era_of(&legacy), Era::Legacy);
229        assert_eq!(
230            era_of(&json!({"jsonrpc": "2.0", "method": "x"})),
231            Era::Legacy
232        );
233    }

324    fn test_tool_header_annotations_invalid() {
325        let bad = |prop: Value| {
326            tool_header_annotations(&json!({"type": "object", "properties": {"p": prop}}))
327                .unwrap_err()
328        };
329        assert!(bad(json!({"type": "string", "x-mcp-header": ""})).contains("valid header"));
330        assert!(bad(json!({"type": "string", "x-mcp-header": "Bad Name"})).contains("valid header"));
331        assert!(bad(json!({"type": "string", "x-mcp-header": "Bad\r\nX"})).contains("valid header"));
332        assert!(bad(json!({"type": "number", "x-mcp-header": "N"})).contains("not a string"));
333        assert!(bad(json!({"type": "object", "x-mcp-header": "O"})).contains("not a string"));
334        assert!(
335            bad(json!({"type": "array", "items": {"type": "string", "x-mcp-header": "I"}}))
336                .contains("reachable")
337        );
338        assert!(
339            bad(json!({"oneOf": [{"type": "string", "x-mcp-header": "One"}]}))
340                .contains("reachable")
341        );
342        let dup = json!({"type": "object", "properties": {
343            "a": {"type": "string", "x-mcp-header": "Tenant"},
344            "b": {"type": "string", "x-mcp-header": "tenant"}

346        assert!(tool_header_annotations(&dup)
347            .unwrap_err()
348            .contains("unique"));
349        let defs =
350            json!({"type": "object", "$defs": {"x": {"type": "string", "x-mcp-header": "D"}}});
351        assert!(tool_header_annotations(&defs)
352            .unwrap_err()
353            .contains("reachable"));
354    }
